\begin{theorem}[Geometric series]
  For $r\neq 1$ and every integer $n\geq 0$,
  \[
    \sum_{k=0}^{n}r^k=\frac{1-r^{n+1}}{1-r}.
  \]
\end{theorem}

\begin{proof}
  Multiply the sum by $1-r$ and cancel adjacent terms.
\end{proof}
